03 缓存层：Redis 原子战场 Lua Script

Running
-- 1. 幂等性检查 (防止重复排队)
if redis.call('sismember', orderSet, uid) == 1 then
return -1 -- 重复请求
end

-- 2. 校验库存 & 原子扣减
local stock = redis.call('get', key)
if tonumber(stock) <= 0 then
return 0 -- 已售罄
else
redis.call('decr', key)
redis.call('sadd', orderSet, uid)
return 1 -- 抢购成功
end
